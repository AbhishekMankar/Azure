from pyspark.sql.functions import *
import config

silver_df = spark.read.format("delta").load(config.SILVER_JOINED_PATH)

dim_product = silver_df.select(
    "product_id",
    "product_name",
    "category",
    "subcategory",
    "brand"
).distinct()

dim_product.write.format("delta").mode("overwrite").save(config.DIM_PRODUCT_PATH)
