from pyspark.sql.functions import *
import config

silver_df = spark.read.format("delta").load(config.SILVER_JOINED_PATH)

dim_store = silver_df.select(
    "store_id",
    "store_name",
    "store_city",
    "store_state"
).distinct()

dim_store.write.format("delta").mode("overwrite").save(config.DIM_STORE_PATH)
