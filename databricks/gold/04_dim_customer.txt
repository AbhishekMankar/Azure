from pyspark.sql.functions import *
import config

silver_df = spark.read.format("delta").load(config.SILVER_JOINED_PATH)

dim_customer = silver_df.select(
    "customer_id",
    "customer_name",
    "customer_type",
    "loyalty_tier"
).distinct()

dim_customer.write.format("delta").mode("overwrite").save(config.DIM_CUSTOMER_PATH)
